\section{Kaluza-klein reduction of the Einstein-Hilbert action}
\begin{frame}{Hilbert--Einstein action}
We are going to follow \cite{scherk1979get} paper for this part.
\begin{block}{The Action}
\begin{equation*}
\begin{aligned}
    S &= \int d x^D \int \frac{d y ^E}{\rho(E)} \sqrt{- \hat g} \hat R  \\
    &=\int d x^D\hat V \{  \hat \omega^{\hat s}{}_{\hat s}{}^{\hat t}\omega^{\hat r}{}_{\hat r \hat t} + \hat \omega^{\hat s \hat t}{}_{ \hat r}\omega^{\hat r}{}_{  \hat s \hat t }+ 2\hat \omega_{\hat s}{}^{\hat r \hat s}\hat V^{\hat \mu}_{ \hat r}\partial_{\hat \mu}\hat \phi \}
\end{aligned}
\end{equation*}
\end{block}
hat $\rightarrow$ high dimension, Begining of an alphabet $\rightarrow$ internal space, end of an alphabet $\rightarrow$ space-time.
\end{frame}
\begin{frame}{Vielbein ansatz}

   \[\hat \omega_{trs } = - \hat\Omega_{\hat t\hat r \hat s} +\hat\Omega_{\hat r\hat s \hat t } -\hat\Omega_{\hat s\hat t \hat r}\]
   \[\hat\Omega_{\hat r\hat s \hat t } = \frac{1}{2}\big(\hat V^{\hat \mu}_{\hat r}\hat V^{\hat \nu}_{\hat s} -\hat V^{\hat \nu}_{\hat r} \hat V^{\hat \mu}_{\hat s}\big)\partial_{\hat \nu} \hat V_{\hat \mu \hat r} \]
 \end{block}

\end{frame}

\section{$O(E,E)$ symmetry}
\begin{frame}{$O(E,E)$ group}
    \begin{definition}
        A matrix $M$ is in $O(E,E)$ if it is a square $E\times E$ matrix and, for the matrix
    \begin{equation*}
        \zeta = \begin{pmatrix}
            0 & 1 \\ 1 &0
        \end{pmatrix},
    \end{equation*}
    which can be used as a metric for the group $O(E,E)$, as it has $E$ negative eigenvalues and $E$ positive eigenvalues in a basis rotated from the one with diagonal metric,\\ 
    the matrix $M$ satisfies
    \begin{equation*}
        \boxed{M^T \zeta M = \zeta. }
    \end{equation*}
    \end{definition}
\end{frame}
\begin{frame}
    \begin{columns}

\begin{block}{$O(E,E)$ group action}
 \begin{align*}
    &A^i \rightarrow \Omega^i_j A^j, \\

\end{align*}
\end{block}
