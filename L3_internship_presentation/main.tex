\begin{frame}
We need a way to describe how a $4$-dimensional observer would see a $4+E$-dimensional universe.\\
\centering\includegraphics[width=0.5\textwidth]{images4.png}\\
\begin{equation*}S_{4+E}= \int dx^{4+E} \mathcal{L}_{4+E} \longrightarrow S_4= \int dx^{4} \mathcal{L}_{4} \end{equation*}
This procedure is called \textbf{dimensional reduction}. Once we've reduced
the action, we can also look for hidden symmetries it develops in its
reduced form symmetries that were invisible in $4+E$ dimensions.

\end{frame}
\section{$4+E$ gravity description}
\begin{frame}{A warm-up: electromagnetism}
    \begin{columns}
        \column{0.4\textwidth}
        \centering{\textbf{Maxwell equations}}
        \begin{itemize}
            \item $\vec{\nabla} \cdot \vec{E} = \mu_0 c^2 \rho$,
            \item $\vec{\nabla} \times \vec{B} = \mu_0 \vec{j} + \frac{1}{c^2} \pdv{\vec{E}}{t}$,
            \item $\vec{\nabla} \times \vec{E} = - \pdv{\vec{B}}{t}$,
            \item $\vec{\nabla} \cdot \vec{B} = 0$.
        \end{itemize}
        \column{0.6\textwidth}\pause
        \centering{\textbf{Covariant formulation}}
        \begin{itemize}
        \item Four-current: $J^\mu = (c \rho , \vec{j})$, \\
        \item Four-potential: $A^\mu = (V/c, \vec{A})$,
        \item Field strength: $F_{\mu \nu} = \partial_\mu A_\nu - \partial_\nu A_\mu$.\\
        \end{itemize}
        \vspace{6mm}\pause
    \end{columns}
    \vspace{10mm}
All four equations are packaged into a single action, the Maxwell action in vacuum:
\begin{equation*}
    \boxed{S_{F} = \int dx^4 \frac{1}{4} F_{\mu \nu}F^{\mu \nu}}
\end{equation*}
\end{frame}
\begin{frame}{4-dimensional gravity}

    \begin{itemize}
     \item General relativity encodes spacetime deformation in a metric:
    \begin{equation*}
        g_{\mu \nu} = \begin{pmatrix}
            g_{00} & g_{01} & g_{02} & g_{03} \\
            g_{01} & g_{11} & g_{12} & g_{13} \\
            g_{02} & g_{12} & g_{22} & g_{23}\\
            g_{03} & g_{13} & g_{23} & g_{33}  
        \end{pmatrix}
    \end{equation*} \pause
    \item  the Hilbert--Einstein action:
    \begin{equation*}
      \boxed{  S_{g} = \int dx^4 \sqrt{-g}\, R}
    \end{equation*}
\end{itemize}\pause
\centering{Einstein's equation: $R_{\mu \nu} +\frac{1}{2}g_{\mu \nu}R = 0$.}
\end{frame}
\begin{frame}{Going to $4+E$ dimensions}
    \begin{itemize}
        \item We enlarge the metric:\\
    \begin{equation*}
           \hat g_{\hat\mu \hat\nu} = \begin{pmatrix}
          4 \times 4  & 4 \times E  \\  E \times 4  & E \times E
        \end{pmatrix}
    \quad \quad \quad \quad \text{ \fontsize{6}{7}\selectfont A little hat mean that we are in high dimension. \normalsize}
    \end{equation*}
    
        \item We assume no field depends on the internal coordinates.
        \item The action is rewritten accordingly:
        \begin{equation*}
            S_{\hat g} = \int dx^4 \int \frac{dy^E}{\rho(E)} \sqrt{- \hat g} \, \hat R
        \end{equation*}
    \end{itemize}
\end{frame}
\begin{frame}{Reduction}
    We make an ansatz on the metric,
    \begin{equation*}
         \hat g_{\hat \mu \hat \nu} = \begin{pmatrix}
        g_{\mu \nu} - A^{(1)\gamma}_{\mu}A^{(1)}_{\nu \lambda} & - A^{(1)}_{\mu \beta} \\  -A^{(1)}_{\nu \alpha} &-h_{\alpha \beta} \end{pmatrix},
    \end{equation*}\pause
    and after some calculation, the reduced action is:
   \begin{equation*}
    \boxed{
    \begin{aligned}
    S_{\hat g} = &\int d x^4\, \sqrt{-g}\Big( \textcolor{red}{R}
    +\textcolor{violet}{\frac{1}{4}F^{\mu \nu \alpha}F_{ \alpha \mu  \nu}}\\
    &-   \frac{1}{4} g^{\rho \lambda} \partial_\lambda h_{\alpha \beta}\partial_\rho h^{\alpha \beta}\Big).
\end{aligned}}
\end{equation*}
\centering{ \textcolor{red}{Low dimension gravity},  \textcolor{violet}{$E$ electromagnetism field}}
\end{frame}
\section{$O(E,E)$ symmetry}
\begin{frame}{Same idea, string theory version: the NS-NS action}
    In string theory we don't use the pure Einstein--Hilbert action, but a variant of it, the NS-NS action:
    \begin{equation*} 
    \boxed{S = \int d x^4 \int \frac{d y^E}{\rho(E)}\sqrt{- \hat g } \, e^{ - \hat \phi} \Big(\hat R +  \partial_{\hat \mu}\hat \phi \, \partial^{\hat \mu}\hat \phi - \frac{1}{12}  H_{ \mu \nu \rho}H^{ \mu\nu \rho}\Big),}
\end{equation*} \pause
\begin{columns}
    \column{0.5\textwidth}
    \centering{\textbf{The two-form $B_{\hat \mu \hat \nu}$}}:\\
    \begin{itemize}
        \item Field strength: $H_{ \mu \nu \rho} = \partial_ \mu B_{\nu \rho} + \text{cyclic perms.}$
        \item Parametrized via the vielbein:
    \begin{align*}
        T^{\mu} &= V_r^\mu \hat V^r_{\hat \mu}\hat T^{\hat \mu},  \quad
        T^{\alpha} = V_a^\alpha \hat V^a_{\hat \mu}\hat T^{\hat \mu}, \\ 
        T_{\mu} &= V_\mu^r \hat V_r^{\hat \mu}\hat T_{\hat \mu}, \quad
        T_{\alpha} = V_\alpha^a \hat V_a^{\hat \mu}\hat T_{\hat \mu}.
    \end{align*}
    \end{itemize}
    \column{0.5\textwidth}
    \centering{\textbf{The dilaton $\hat \phi$}}:\\
    \begin{itemize}
        \item Shifted dilaton: $\phi = \hat \phi -\frac{1}{2} \log{h}$
        \item $\partial_\alpha \hat \phi = 0 $
    \end{itemize}
    \vspace{16mm}
\end{columns}
\end{frame}
\begin{frame}{The reduced action}
    Carrying out the reduction, we obtain:
    \begin{equation*}
        \boxed{
    \begin{aligned}
    S = &\int  dx^4\, Ve^{-\phi}\Big( \textcolor{red}{R} +\textcolor{blue}{\partial_{\mu }\phi \, \partial^{\mu} \phi }+ \frac{1}{8}\tr(\partial_\mu M^{-1}\partial^\mu M)\\ 
    & \textcolor{violet}{-\frac{1}{4} \mathcal{F}^i_{\mu \nu}(M^{-1})_{ij} \mathcal{F}^{\mu \nu j}}\textcolor{green}{-\frac{1}{12}H_{\mu \nu \rho} H^{\mu \nu \rho}} \Big),
    \end{aligned}}
\end{equation*}
 \centering{ \textcolor{red}{Low dimension gravity},  \textcolor{violet}{$E$ modified electromagnetism field},   \textcolor{blue}{Shifted dilaton field}, \textcolor{green}{low dimension two form field}   }\\
with $M$ built from the internal metric $h$ and the two-form $B$:
\begin{equation*}
     M = \begin{pmatrix}
        h^{-1} & -h^{-1}B \\ Bh^{-1} & h-Bh^{-1}B
    \end{pmatrix}.
\end{equation*}

\end{frame}
\begin{frame}{$O(E,E)$ invariance}
    We define the action of the $O(E,E)$ group as
    \begin{align*} 
    &A^i \rightarrow \Omega^i_j A^j, \\
    &M \rightarrow \Omega^T M \, \Omega,
\end{align*}
where $\Omega \in O(E,E)$. \\
\vspace{4mm}
\centering{\textbf{...the action is left completely invariant!}}
\end{frame}

\begin{frame}{Definition of $O(E,E)$}
     A square $E\times E$ matrix $M$ belongs to $O(E,E)$ if, for the matrix
    \begin{equation*}
        \zeta = \begin{pmatrix}
            0 & 1 \\ 1 &0
        \end{pmatrix},
    \end{equation*}
    which can be used as a metric for the group $O(E,E)$ since $\zeta$ has $E$ negative and $E$ positive eigenvalues in a basis rotated from the diagonal one $M$ satisfies
    \begin{equation*}
        \boxed{M^T \zeta M = \zeta. }
    \end{equation*}
\end{frame}
